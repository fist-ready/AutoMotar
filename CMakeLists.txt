cmake_minimum_required(VERSION 3.20)
project(AutoMortar VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(measurement_test tests/measurement_test.cpp)
target_include_directories(measurement_test PRIVATE src)
add_test(NAME measurement COMMAND measurement_test)
if(WIN32)
    enable_language(RC)
    add_executable(AutoMortar WIN32 src/main.cpp src/AutoMortar.rc)
    target_compile_definitions(AutoMortar PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(AutoMortar PRIVATE user32 gdi32 shell32)
    if(MSVC)
        target_link_options(AutoMortar PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/AutoMortar.manifest")
    else()
        target_link_options(AutoMortar PRIVATE -municode -static-libgcc -static-libstdc++)
    endif()
    set_target_properties(AutoMortar PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
    add_test(NAME windows_smoke COMMAND AutoMortar --smoke-test)
    set_tests_properties(windows_smoke PROPERTIES TIMEOUT 20)
endif()
if(MSVC)
    foreach(target measurement_test AutoMortar)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
else()
    target_compile_options(measurement_test PRIVATE -Wall -Wextra -pedantic)
endif()
